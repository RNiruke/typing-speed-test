<script>

function toggleIssueMenu() {
    let menu = document.getElementById("issueMenu");

    menu.style.display =
        menu.style.display === "block" ? "none" : "block";
}

function selectIssue(element) {

    let index = element.dataset.index;

    let allIssues =
        document.querySelectorAll("#hiddenIssues .issue-data");

    let display =
        document.getElementById("issueDisplay");

    // show ONLY selected issue
    display.innerHTML = allIssues[index].innerHTML;

    // close dropdown
    document.getElementById("issueMenu").style.display = "none";
}

</script>
